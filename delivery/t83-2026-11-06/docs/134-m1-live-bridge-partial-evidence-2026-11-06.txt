    99	    """Sum token usage recorded for one job.
   100	
   101	    Rows may be bound by ``job_id`` directly (assistant calls) or by
   102	    ``attempt_id`` (route calls), so both paths are matched.  The ``OR`` cannot
   103	    double count a row: each row is selected at most once.
   104	
   105	    Semantics:
   106	      * no calls                 -> None
   107	      * calls but no real tokens -> None
   108	      * some NULL, some real     -> sum the real ones, ``partial`` is True
   109	      * all real                 -> sum, ``partial`` is False
   110	
   111	    A NULL count is never read as zero, and one incomplete row does not erase
   112	    the real usage its siblings recorded -- it only marks the sum partial.
   113	    """
   114	    connection = connect(data_dir)
   115	    try:
   116	        rows = connection.execute(
   117	            "SELECT prompt_tokens, completion_tokens FROM model_calls "
   118	            "WHERE job_id = ? OR attempt_id IN ("
   119	            "  SELECT a.id FROM attempts a JOIN runs r ON r.id = a.run_id "
   120	            "  WHERE r.job_id = ?)",
   121	            (job_id, job_id),
   122	        ).fetchall()
   123	    finally:
   124	        connection.close()
   125	    if not rows:
   126	        return None
   127	    real = [
   128	        row for row in rows
   129	        if row["prompt_tokens"] is not None and row["completion_tokens"] is not None
   130	    ]
   131	    if not real:
   132	        return None
   133	    return {
   134	        "input_tokens": sum(int(row["prompt_tokens"]) for row in real),
   135	        "output_tokens": sum(int(row["completion_tokens"]) for row in real),
   136	        "partial": len(real) != len(rows),
   137	    }
